    ```ts
    import { z } from 'zod';

    /**
     * D-24 / DEPLOY-03: server fails fast at startup with a single readable line
     * per missing required env var.
     *
     * Required vars: DATABASE_URL, LINEAR_CLIENT_ID, LINEAR_CLIENT_SECRET, LINEAR_WEBHOOK_SECRET.
     * AGENTWATCH_INTERNAL_API_KEY is required so the web container can authenticate to the server.
     * WORKSPACE_ID is optional at boot (set after setup wizard runs).
     */
    const EnvSchema = z.object({
      DATABASE_URL: z.string().url(),
      LINEAR_CLIENT_ID: z.string().min(1),
      LINEAR_CLIENT_SECRET: z.string().min(1),
      LINEAR_WEBHOOK_SECRET: z.string().min(1),
      AGENTWATCH_INTERNAL_API_KEY: z.string().min(16),

      // Set by setup wizard, not at first boot
      WORKSPACE_ID: z.string().uuid().optional(),

      // Optional with sensible defaults
      IDENTITY_CONFIDENCE_THRESHOLD: z.coerce.number().min(0).max(1).default(0.8),
      LOG_LEVEL: z.enum(['debug', 'info', 'warn', 'error']).default('info'),
      TELEMETRY_OPT_IN: z
        .union([z.literal('true'), z.literal('false')])
        .default('false')
        .transform((v) => v === 'true'),
      PORT: z.coerce.number().int().min(1).max(65535).default(8080),
    });
    export type Env = z.infer<typeof EnvSchema>;

    /**
     * Parse env, fail-fast with one readable line per error.
     * In tests, we re-throw instead of calling process.exit so vitest can assert.
     */
    export function loadEnv(source: NodeJS.ProcessEnv = process.env, opts: { exitOnError?: boolean } = { exitOnError: true }): Env {
      const result = EnvSchema.safeParse(source);
      if (!result.success) {
        const lines: string[] = [];
        for (const issue of result.error.issues) {
          const key = issue.path.join('.');
          lines.push(`FATAL: missing env ${key}: ${issue.message}`);
        }
        for (const line of lines) console.error(line);
        if (opts.exitOnError) process.exit(1);
        throw new Error(`env validation failed:\n${lines.join('\n')}`);
      }
      return result.data;
    }

    /** D-30 boot log: print telemetry status so users can verify it is off (Pitfall 7 P1 prep). */
    export function logBootBanner(env: Env, log: { info: (msg: string) => void }) {
      log.info(`agentwatch starting on port ${env.PORT}`);
      log.info(`telemetry: ${env.TELEMETRY_OPT_IN ? 'on' : 'off'}`);
      log.info(`identity_confidence_threshold: ${env.IDENTITY_CONFIDENCE_THRESHOLD}`);
      log.info(`log_level: ${env.LOG_LEVEL}`);
    }
    ```

    `packages/server/src/plugins/request-logger.ts` (D-29):
    ```ts
    import type { FastifyReply, FastifyRequest } from 'fastify';

    /**
     * D-29: Webhook log line fields ONLY: delivery_id, source, event_type, bytes_received, latency_ms.
     * At LOG_LEVEL=debug add payload_keys (top-level keys, never values).
     * NEVER log req.body, headers, or signatures (Pitfall 5, 12).
     */
    export function logWebhookReceipt(
      req: FastifyRequest,
      reply: FastifyReply,
      source: 'linear' | 'github' | 'sdk',
    ) {
      const start = process.hrtime.bigint();
      reply.raw.on('finish', () => {
        const latencyMs = Number((process.hrtime.bigint() - start) / 1_000_000n);
        const deliveryId =
          (req.headers['linear-delivery'] as string | undefined) ??
          (req.headers['x-github-delivery'] as string | undefined) ??
          null;
        // event_type is read from the parsed body — top-level only, never deep paths.
        const body = req.body as Record<string, unknown> | undefined;
        const eventType =
          (body && typeof body['type'] === 'string' && body['type']) ||
          (body && typeof body['event_type'] === 'string' && body['event_type']) ||
          null;
        const bytes = req.rawBody?.length ?? 0;

        const fields: Record<string, unknown> = {
          delivery_id: deliveryId,
          source,
          event_type: eventType,
          bytes_received: bytes,
          latency_ms: latencyMs,
        };
        if (process.env.LOG_LEVEL === 'debug' && body && typeof body === 'object') {
          fields.payload_keys = Object.keys(body);  // top-level keys only — D-29
        }
        req.log.info(fields, 'webhook_received');
      });
    }
    ```

    `packages/server/src/plugins/auth.ts` (RESEARCH Pattern 5):
    ```ts
    import fp from 'fastify-plugin';
    import { createHash, timingSafeEqual } from 'node:crypto';
    import type { FastifyReply, FastifyRequest } from 'fastify';
    import { sql } from 'drizzle-orm';
    import type { ServerDb } from '../db';

    declare module 'fastify' {
      interface FastifyRequest {
        workspaceId?: string;
      }
      interface FastifyInstance {
        authBearer: (req: FastifyRequest, reply: FastifyReply) => Promise<void>;
      }
    }

    export default fp<{ db: ServerDb }>(async (fastify, opts) => {
      fastify.decorate('authBearer', async (req: FastifyRequest, reply: FastifyReply) => {
        const auth = req.headers.authorization;
        if (typeof auth !== 'string' || !auth.startsWith('Bearer ')) {
          return reply.code(401).send({ error: 'missing_bearer' });
        }
        const presented = auth.slice('Bearer '.length).trim();
        if (presented.length === 0) {
          return reply.code(401).send({ error: 'missing_bearer' });
        }
        const presentedHash = createHash('sha256').update(presented).digest();

        // Single-tenant in P1: take the only workspaces row.
        const r = await opts.db.execute(sql`
          SELECT id, api_key_hash FROM workspaces LIMIT 1
        `);
        const ws = r.rows[0] as { id: string; api_key_hash: string } | undefined;
        if (!ws) {
          return reply.code(401).send({ error: 'no_workspace' });
        }
        const stored = Buffer.from(ws.api_key_hash, 'hex');
        if (presentedHash.length !== stored.length || !timingSafeEqual(presentedHash, stored)) {
          return reply.code(401).send({ error: 'unauthorized' });
        }
        req.workspaceId = ws.id;
      });
    });
    ```

    ```ts
    import Fastify from 'fastify';
    import pino from 'pino';
    import { migrate } from 'drizzle-orm/node-postgres/migrator';
    import { fileURLToPath } from 'node:url';
    import { dirname, resolve } from 'node:path';
    import { loadEnv, logBootBanner } from './env';
    import { createServerDb } from './db';
    import rawBody from './plugins/raw-body';
    import metricsPlugin, { registerMetricsRoute } from './plugins/metrics';
    import authPlugin from './plugins/auth';
    import healthRoutes from './routes/health';

    async function main() {
      const env = loadEnv();

      const logger = pino({
        level: env.LOG_LEVEL,
        redact: {
          paths: [
            'req.body',
            'req.headers.authorization',
            'req.headers["linear-signature"]',
            'req.headers["x-hub-signature-256"]',
            'res.body',
            '*.payload',
            '*.signature',
            'workspace_salt',
            'api_key_hash',
          ],
          censor: '[REDACTED]',
          remove: false,
        },
        serializers: {
          req: (r) => ({ id: r.id, method: r.method, url: r.url }),
          res: (r) => ({ statusCode: r.statusCode }),
        },
      });

      logBootBanner(env, logger);

      // === BLOCKING: migrations apply BEFORE server accepts connections (D-05) ===
      const { db, pool } = createServerDb(env.DATABASE_URL);
      // Migrations directory: packages/db/migrations relative to monorepo root
      // Server is built into packages/server/dist/, so migrations live at ../../db/migrations
      const here = typeof __dirname !== 'undefined' ? __dirname : dirname(fileURLToPath(import.meta.url));
      const migrationsFolder = resolve(here, '..', '..', 'db', 'migrations');
      try {
        logger.info({ migrationsFolder }, 'running migrations');
        await migrate(db, { migrationsFolder });
        logger.info('migrations applied');
      } catch (err) {
        logger.fatal({ err }, 'FATAL: migration failed at startup');
        process.exit(1);
      }
      // === END BLOCKING ===

      const fastify = Fastify({
        loggerInstance: logger,
        disableRequestLogging: true,  // Pitfall 12: we emit our own webhook log lines from Wave 2
      });
      fastify.decorate('env', env);
      fastify.decorate('db', db);
      fastify.decorate('pgPool', pool);

      await fastify.register(rawBody);
      await fastify.register(metricsPlugin, { db });
      await fastify.register(authPlugin, { db });
      await fastify.register(healthRoutes, { db });
      await registerMetricsRoute(fastify);

      // Wave 2 will register /webhooks/linear, /api/v1/query, /api/v1/sdk/event,
      // /api/v1/agents/[id]/confirm here.

      fastify.addHook('onClose', async () => { await pool.end(); });

      await fastify.listen({ port: env.PORT, host: '0.0.0.0' });
      logger.info(`listening on :${env.PORT}`);
    }

    main().catch((err) => {
      console.error('FATAL:', err);
      process.exit(1);
    });
    ```

|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Information Disclosure | `req.body` reaches log aggregator via accidental `req.log.info(req)` | mitigate | Pino `redact.paths` includes `req.body`, `req.headers.authorization`, `req.headers["linear-signature"]`, `req.headers["x-hub-signature-256"]`.
