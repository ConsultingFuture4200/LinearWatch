    ```ts
    import type { FastifyInstance } from 'fastify';
    import { createHmac, timingSafeEqual } from 'node:crypto';
    import { sql } from 'drizzle-orm';
    import { logWebhookReceipt } from '../../plugins/request-logger';
    import { eventsReceived, webhookAckSeconds } from '../../plugins/metrics';

    const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

    /**
     * D-04 (Pitfall 2): handler does exactly four things synchronously:
     *   1. Validate Linear-Delivery (400 on missing/malformed — D-09)
     *   2. HMAC verify against LINEAR_WEBHOOK_SECRET (401 on failure)
     *   3. INSERT ... ON CONFLICT DO NOTHING into events.raw_event
     *      using req.rawBody bytes cast to jsonb in SQL ($N::text)::jsonb
     *      (so the hashed bytes and the stored bytes are byte-identical)
     *   4. Enqueue Graphile Worker resolve_identity job (only if new insert)
     * Returns 200. NO DB joins, vendor calls, or resolver runs here.
     *
     * D-20: webhooks are HMAC-authenticated, NOT Bearer-authenticated.
     * OBS-04: this file MUST NOT reference req.body — only req.rawBody.
     */
    export default async function linearWebhookRoute(fastify: FastifyInstance) {
      fastify.post('/webhooks/linear', async (req, reply) => {
        logWebhookReceipt(req, reply, 'linear');
        const stop = webhookAckSeconds.startTimer({ source: 'linear' });
        try {
          // === 1. Linear-Delivery validation (D-09) ===
          const deliveryRaw = req.headers['linear-delivery'];
          const deliveryId = typeof deliveryRaw === 'string' ? deliveryRaw : Array.isArray(deliveryRaw) ? deliveryRaw[0] : undefined;
          if (!deliveryId || !UUID_RE.test(deliveryId)) {
            reply.code(400);
            return { error: 'missing_or_invalid_linear_delivery' };
          }

          // === 2. HMAC verify (INGEST-01) ===
          const sigRaw = req.headers['linear-signature'];
          const signature = typeof sigRaw === 'string' ? sigRaw : Array.isArray(sigRaw) ? sigRaw[0] : undefined;
          if (!signature || !req.rawBody) {
            reply.code(401);
            return { error: 'missing_signature' };
          }
          // Linear's signature is hex(HMAC-SHA256(secret, rawBody))
          const expected = createHmac('sha256', fastify.env.LINEAR_WEBHOOK_SECRET)
            .update(req.rawBody)
            .digest('hex');
          let valid = false;
          try {
            const sigBuf = Buffer.from(signature, 'hex');
            const expBuf = Buffer.from(expected, 'hex');
            if (sigBuf.length === expBuf.length) {
              valid = timingSafeEqual(sigBuf, expBuf);
            }
          } catch {
            valid = false;
          }
          if (!valid) {
            reply.code(401);
            return { error: 'invalid_signature' };
          }

          // === 3. INSERT idempotently (INGEST-05, Pitfall 1) ===
          // The raw bytes (UTF-8 string of req.rawBody) are bound as a TEXT parameter
          // and cast to jsonb inside SQL — guarantees byte-identity between hashed and stored payload.
          // RETURNING (xmax = 0) tells us if the row was newly inserted (true) or pre-existed (false).
          const rawBodyText = req.rawBody.toString('utf8');
          const result = await fastify.db.execute(sql`
            INSERT INTO events.raw_event (source, upstream_id, payload, signature_valid)
            VALUES ('linear', ${deliveryId}, (${rawBodyText}::text)::jsonb, true)
            ON CONFLICT (source, upstream_id, received_at) DO NOTHING
            RETURNING id, (xmax = 0) AS is_new
          `);
          const row = (result.rows as Array<{ id: string; is_new: boolean }>)[0];
          eventsReceived.inc({ source: 'linear' });

          // === 4. Enqueue resolver job ONLY for new inserts ===
          if (row && row.is_new) {
            // Single-tenant in P1: pull workspaces.id (cached after first lookup).
            const ws = await getOrCacheWorkspaceId(fastify);
            if (ws) {
              await fastify.db.execute(sql`
                SELECT graphile_worker.add_job('resolve_identity', ${JSON.stringify({
                  raw_event_id: row.id,
                  workspace_id: ws,
                })}::jsonb)
              `);
            }
          }

          // === 5. Always 200 (Pitfall 1: never let Linear retry once we've durably stored or de-duped) ===
          reply.code(200);
          return { ok: true };
        } finally {
          stop();
        }
      });
    }

    let workspaceIdCache: string | null = null;
    async function getOrCacheWorkspaceId(fastify: FastifyInstance): Promise<string | null> {
      if (workspaceIdCache) return workspaceIdCache;
      const r = await fastify.db.execute(sql`SELECT id FROM workspaces LIMIT 1`);
      const id = (r.rows as Array<{ id: string }>)[0]?.id ?? null;
      workspaceIdCache = id;
      return id;
    }
    ```

    ```ts
      const admin = new Pool({ connectionString: TEST_DB_URL.replace(/\/[^/]+$/, '/postgres') });
      await admin.query('DROP DATABASE IF EXISTS linearwatch_bench');
      await admin.query('CREATE DATABASE linearwatch_bench');
      await admin.end();

      // 2. Start the server
      const proc = spawn('node', ['packages/server/dist/index.js'], {
        env: {
          ...process.env,
          DATABASE_URL: TEST_DB_URL,
          LINEAR_CLIENT_ID: 'cid', LINEAR_CLIENT_SECRET: 'sec',
          LINEAR_WEBHOOK_SECRET: SECRET,
          LINEARWATCH_INTERNAL_API_KEY: 'bench-internal-key-1234567890',
          PORT: String(PORT),
        },
        stdio: 'pipe',
      });
      proc.stdout?.on('data', () => {});
      proc.stderr?.on('data', (d) => process.stderr.write(d));

      // 3. Wait for /health
      const start = Date.now();
      let ready = false;
      while (Date.now() - start < 30_000) {
        try {
          const r = await fetch(`http://127.0.0.1:${PORT}/health`);
          if (r.ok) { ready = true; break; }
        } catch {}
        await new Promise((r) => setTimeout(r, 250));
      }
      if (!ready) {
        proc.kill('SIGTERM');
        throw new Error('server not ready');
      }

      // 4. Seed the workspace
      const pool = new Pool({ connectionString: TEST_DB_URL });
      await pool.query(`
        INSERT INTO workspaces (id, name, api_key_hash, workspace_salt)
        VALUES ('00000000-0000-4000-8000-000000000001', 'bench', 'placeholder-hash', 'salt')
      `);
      await pool.end();

      // 5. Fire 200 concurrent unique requests
      // We pre-generate 1000 unique signed payloads so autocannon doesn't repeat the same delivery
      // (which would cause ON CONFLICT short-circuits and skew the latency distribution upward).
      const payloads: { url: string; method: 'POST'; headers: Record<string, string>; body: string }[] = [];
      for (let i = 0; i < 1000; i++) {
        const body = JSON.stringify({ type: 'AgentSession', actor: { id: `lap-${i}` }, n: i });
        const sig = createHmac('sha256', SECRET).update(body).digest('hex');
        payloads.push({
          url: `http://127.0.0.1:${PORT}/webhooks/linear`,
          method: 'POST',
          headers: {
            'content-type': 'application/json',
            'linear-delivery': randomUUID(),
            'linear-signature': sig,
          },
          body,
        });
      }
      let cursor = 0;

      const result = await autocannon({
        url: `http://127.0.0.1:${PORT}/webhooks/linear`,
        connections: 200,            // D-31: 200 concurrent
        duration: 15,                 // 15s soak
        requests: payloads.map((p) => ({
          method: p.method,
          path: '/webhooks/linear',
          headers: p.headers,
          body: p.body,
        })),
      });

      proc.kill('SIGTERM');
      // 6. Assert p99 < 200ms (D-31)
      // autocannon reports latency in ms.
      const p99 = result.latency.p99;
      console.log(JSON.stringify({ p50: result.latency.p50, p99, p999: result.latency.p99_9, errors: result.errors }, null, 2));
      if (p99 >= 200) {
        console.error(`FAIL: p99 ack latency ${p99}ms >= 200ms SLA (D-31)`);
        process.exit(1);
      }
      if (result.non2xx > 0) {
        console.error(`FAIL: ${result.non2xx} non-2xx responses during benchmark`);
        process.exit(1);
      }
      console.log(`OK: p99=${p99}ms (< 200ms)`);
      process.exit(0);
    }

    main().catch((err) => { console.error(err); process.exit(1); });
    ```
