    ```ts
    import type { Window } from '@agentwatch/shared';

    /** Convert a Window into a (sinceDate, untilDate) pair for SQL parameterization. */
    export function resolveWindow(w: Window): { since: Date; until: Date } {
      const now = new Date();
      if (w.last) {
        const m = /^(\d+)(d|h)$/.exec(w.last);
        if (!m) throw new Error('invalid_window_last');
        const n = parseInt(m[1]!, 10);
        const ms = m[2] === 'd' ? n * 86_400_000 : n * 3_600_000;
        return { since: new Date(now.getTime() - ms), until: now };
      }
      return { since: new Date(w.from!), until: new Date(w.to!) };
    }
    ```

    ```ts
    import type { QueryRequest, QueryRow, DimensionName } from '@agentwatch/shared';
    import type { ServerDb } from '../../db';
    import { resolveWindow } from '../window';

    /**
     * cost_by_agent: SUM(cost_usd), COUNT(*) grouped by dimension within window+filters.
     *
     * Each metric file is a STATIC SQL function (API-03). Filters are applied via
     * parameterized predicates only — `field` and `op` are Zod enums (FilterField,
     * FilterOp), so no string concatenation enters the SQL text from user input.
     */
    export async function costByAgent(
      req: QueryRequest,
      ctx: { workspaceId: string; db: ServerDb },
    ): Promise<QueryRow[]> {
      const { since, until } = resolveWindow(req.window);
      const dim: DimensionName = req.dimension ?? 'agent';

      const dimSql = dimColumnSql(dim);    // closed-set lookup
      const dimJoin = dimJoinSql(dim);
      const dimKey = dimKeyExprSql(dim);
      const filterSql = buildFilterSql(req.filters);

      // The SQL is built from CLOSED-SET lookups only. Workspace + window are bound parameters.
      const text = `
        SELECT ${dimKey} AS key,
               COALESCE(SUM(s.cost_usd), 0)::float8 AS value,
               COUNT(*)::int AS count
          FROM agent_sessions s
          ${dimJoin}
         WHERE s.workspace_id = $1
           AND s.started_at >= $2
           AND s.started_at <  $3
           ${filterSql.text}
         GROUP BY ${dimSql}
         ORDER BY value DESC
         LIMIT 200
      `;
      const params = [ctx.workspaceId, since, until, ...filterSql.params];
      const r = await ctx.db.execute({ text, args: params } as any).catch(async () => {
        // Drizzle's execute interface varies; fall back to underlying pool if necessary.
        // The executor should pick whichever signature works for the active Drizzle version (0.45.x).
        const pool = (ctx.db as any).$client ?? (ctx.db as any)._pool;
        return pool.query({ text, values: params });
      });
      return (r.rows as Array<{ key: string; value: number; count: number }>).map(row => ({
        key: row.key,
        value: Number(row.value),
        count: Number(row.count),
      }));
    }

    function dimColumnSql(dim: DimensionName): string {
      switch (dim) {
        case 'agent': return 'a.id, a.name';
        case 'team':  return 't.id, t.key, t.name';
        case 'cycle': return 'c.id, c.name';
      }
    }
    function dimJoinSql(dim: DimensionName): string {
      switch (dim) {
        case 'agent': return 'JOIN agents a ON a.id = s.agent_id AND a.deleted_at IS NULL';
        case 'team':  return 'JOIN teams  t ON t.id = s.team_id';
        case 'cycle': return 'JOIN cycles c ON c.id = s.cycle_id';
      }
    }
    function dimKeyExprSql(dim: DimensionName): string {
      switch (dim) {
        case 'agent': return 'a.name';
        case 'team':  return 't.key';
        case 'cycle': return 'c.name';
      }
    }
    function buildFilterSql(filters?: Array<{ field: 'agent_id'|'team_id'|'cycle_id'; op: 'eq'|'in'|'neq'; value: string|string[] }>) {
      if (!filters || filters.length === 0) return { text: '', params: [] as unknown[] };
      const parts: string[] = [];
      const params: unknown[] = [];
      let i = 4; // $1=workspace, $2=since, $3=until
      for (const f of filters) {
        if (f.op === 'eq') {
          parts.push(`AND s.${f.field} = $${i}`);
          params.push(f.value);
          i++;
        } else if (f.op === 'neq') {
          parts.push(`AND s.${f.field} <> $${i}`);
          params.push(f.value);
          i++;
        } else if (f.op === 'in') {
          const arr = Array.isArray(f.value) ? f.value : [f.value];
          parts.push(`AND s.${f.field} = ANY($${i}::uuid[])`);
          params.push(arr);
          i++;
        }
      }
      return { text: parts.join(' '), params };
    }
    ```

    ```ts
    import type { QueryRequest, QueryRow, DimensionName } from '@agentwatch/shared';
    import type { ServerDb } from '../../db';
    import { resolveWindow } from '../window';

    export async function agentSessionCount(
      req: QueryRequest,
      ctx: { workspaceId: string; db: ServerDb },
    ): Promise<QueryRow[]> {
      const { since, until } = resolveWindow(req.window);
      if (!req.dimension) {
        // Aggregate count over the whole window — single row keyed 'all'.
        const text = `
          SELECT 'all' AS key, COUNT(*)::int AS value
            FROM agent_sessions s
           WHERE s.workspace_id=$1 AND s.started_at >= $2 AND s.started_at < $3
        `;
        const r = await runQuery(ctx, text, [ctx.workspaceId, since, until]);
        return r.map(x => ({ key: x.key, value: Number(x.value) }));
      }
      // Reuse cost_by_agent's grouping shape; just swap the value to COUNT(*)
      const dim = req.dimension;
      const join = dim === 'agent'
        ? 'JOIN agents a ON a.id=s.agent_id AND a.deleted_at IS NULL'
        : dim === 'team'
        ? 'JOIN teams t ON t.id=s.team_id'
        : 'JOIN cycles c ON c.id=s.cycle_id';
      const keyExpr = dim === 'agent' ? 'a.name' : dim === 'team' ? 't.key' : 'c.name';
      const groupCol = dim === 'agent' ? 'a.id, a.name' : dim === 'team' ? 't.id, t.key' : 'c.id, c.name';
      const text = `
        SELECT ${keyExpr} AS key, COUNT(*)::int AS value
          FROM agent_sessions s
          ${join}
         WHERE s.workspace_id=$1 AND s.started_at >= $2 AND s.started_at < $3
         GROUP BY ${groupCol}
         ORDER BY value DESC
         LIMIT 200
      `;
      const r = await runQuery(ctx, text, [ctx.workspaceId, since, until]);
      return r.map(x => ({ key: x.key, value: Number(x.value) }));
    }

    async function runQuery(ctx: { db: ServerDb }, text: string, params: unknown[]) {
      const pool = (ctx.db as any).$client ?? (ctx.db as any)._pool;
      const r = await pool.query({ text, values: params });
      return r.rows as Array<{ key: string; value: number }>;
    }
    ```

    ```ts
    import type { QueryRequest, QueryRow, MetricName } from '@agentwatch/shared';
    import type { ServerDb } from '../db';
    import { costByAgent } from './metrics/cost-by-agent';
    import { agentSessionCount } from './metrics/agent-session-count';

    type SqlFn = (req: QueryRequest, ctx: { workspaceId: string; db: ServerDb }) => Promise<QueryRow[]>;

    /**
     * API-03: each metric maps to a STATIC SQL function. Adding a metric is a
     * one-line change here AND a new file in src/query/metrics/. Unknown values
     * are rejected by Zod parsing in the route handler before reaching this map.
     */
    const handlers: Record<MetricName, SqlFn> = {
      cost_by_agent: costByAgent,
      agent_session_count: agentSessionCount,
    };

    export async function dispatchQuery(
      req: QueryRequest,
      ctx: { workspaceId: string; db: ServerDb },
    ): Promise<QueryRow[]> {
      const fn = handlers[req.metric];
      if (!fn) throw new Error('unknown_metric');  // unreachable when Zod parses first
      return fn(req, ctx);
    }
    ```

    ```ts
    import type { FastifyInstance } from 'fastify';
    import { QueryRequest } from '@agentwatch/shared';
    import { dispatchQuery } from '../../../query/dispatcher';

    export default async function queryRoute(fastify: FastifyInstance) {
      fastify.post(
        '/api/v1/query',
        { preHandler: fastify.authBearer },
        async (req, reply) => {
          const parsed = QueryRequest.safeParse(req.body);
          if (!parsed.success) {
            reply.code(400);
            return { error: 'invalid_request', details: parsed.error.format() };
          }
          const wsId = req.workspaceId!;
          const rows = await dispatchQuery(parsed.data, { workspaceId: wsId, db: fastify.db });
          return {
            metric: parsed.data.metric,
            dimension: parsed.data.dimension,
            window: parsed.data.window,
            rows,
          };
        },
      );
    }
    ```

    ```ts
    import { createHash } from 'node:crypto';
    import { sql } from 'drizzle-orm';
    import { SdkEventBody } from '@agentwatch/shared';
    import { logWebhookReceipt } from '../../../plugins/request-logger';
    import { eventsReceived, webhookAckSeconds } from '../../../plugins/metrics';

    /**
     * INGEST-03 + D-10 + D-19. Bearer-authenticated SDK ingestion endpoint.
     * Idempotency key: caller-supplied OR synthesized server-side.
     */
    export default async function sdkEventRoute(fastify: FastifyInstance) {
      fastify.post(
        '/api/v1/sdk/event',
        { preHandler: fastify.authBearer },
        async (req, reply) => {
          logWebhookReceipt(req, reply, 'sdk');
          const stop = webhookAckSeconds.startTimer({ source: 'sdk' });
          try {
            const parsed = SdkEventBody.safeParse(req.body);
            if (!parsed.success) {
              reply.code(400);
              return { error: 'invalid_event', details: parsed.error.format() };
            }
            const ev = parsed.data;
            const wsId = req.workspaceId!;

            const idempotencyKey = ev.idempotency_key
              ?? synthesizeKey(wsId, ev.session_id, ev.event_type, ev.occurred_at);

            await fastify.db.execute(sql`
              INSERT INTO events.raw_event (source, upstream_id, payload, signature_valid)
              VALUES ('sdk', ${idempotencyKey}, ${parsed.data as any}::jsonb, true)
              ON CONFLICT (source, upstream_id, received_at) DO NOTHING
            `);
            eventsReceived.inc({ source: 'sdk' });
            return { ok: true, idempotency_key: idempotencyKey };
          } finally {
            stop();
          }
        },
      );
    }

    function synthesizeKey(workspaceId: string, sessionId: string, eventType: string, occurredAt: string): string {
      const minuteBucket = new Date(occurredAt).toISOString().slice(0, 16);  // 'YYYY-MM-DDTHH:MM'
      return createHash('sha256')
        .update(`${workspaceId}|${sessionId}|${eventType}|${minuteBucket}`)
        .digest('hex');
    }
    ```
