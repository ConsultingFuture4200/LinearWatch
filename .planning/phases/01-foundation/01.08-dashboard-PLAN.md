    ```ts
    import type { QueryRequest, QueryResponse } from '@linearwatch/shared';

    /**
     * The single typed client for POST /api/v1/query.
     * No React component should bypass this; lint guard in CI verifies no
     * `from 'pg'` or `from 'drizzle-orm'` imports under packages/web/src/.
     */
    export async function fetchQuery(req: QueryRequest): Promise<QueryResponse> {
      const url = `${process.env.LINEARWATCH_INTERNAL_URL}/api/v1/query`;
      const r = await fetch(url, {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          authorization: `Bearer ${process.env.LINEARWATCH_INTERNAL_API_KEY}`,
        },
        body: JSON.stringify(req),
        cache: 'no-store',
      });
      if (!r.ok) {
        const text = await r.text();
        throw new Error(`query failed: ${r.status} ${text}`);
      }
      return r.json();
    }
    ```

    ```ts
    /** Confirm action used by the identity side panel. Server-side only. */
    export async function confirmAgent(agentId: string, linearAppUserId: string) {
      const url = `${process.env.LINEARWATCH_INTERNAL_URL}/api/v1/agents/${encodeURIComponent(agentId)}/confirm`;
      const r = await fetch(url, {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          authorization: `Bearer ${process.env.LINEARWATCH_INTERNAL_API_KEY}`,
        },
        body: JSON.stringify({ linear_app_user_id: linearAppUserId }),
        cache: 'no-store',
      });
      if (!r.ok) throw new Error(`confirm failed: ${r.status}`);
      return r.json() as Promise<{ ok: boolean; confirmed_count: number }>;
    }

    /** Fetch active workspace warnings (server-side; called by the dashboard layout). */
    export async function fetchActiveWarnings(): Promise<Array<{ id: string; severity: string; message: string }>> {
      // P1 simplification: no dedicated API yet; rely on the query API's future surface.
      // For P1, use a small purpose-built endpoint added to the server in 01.07's stretch,
      // OR (acceptable shortcut) read from the server via a path proxy. Pick whichever the
      // executor prefers; document in the summary. Default below uses an explicit endpoint.
      const url = `${process.env.LINEARWATCH_INTERNAL_URL}/api/v1/workspace/warnings`;
      const r = await fetch(url, {
        headers: { authorization: `Bearer ${process.env.LINEARWATCH_INTERNAL_API_KEY}` },
        cache: 'no-store',
      });
      if (!r.ok) return [];
      const json = await r.json();
      return json.warnings ?? [];
    }
    ```

    ```tsx
    import { fetchQuery } from '@/lib/query';
    import { CostChart } from '@/components/cost-chart';
    import { AgentsTable } from '@/components/agents-table';
    import { EmptyState } from '@/components/empty-state';

    type SearchParams = { team?: string; cycle?: string; window?: '7d' | '14d' | '30d' | '90d' };

    export default async function CostPage({ searchParams }: { searchParams: Promise<SearchParams> }) {
      const sp = await searchParams;
      const window = (sp.window ?? '14d') as SearchParams['window'];

      const filters: any[] = [];
      if (sp.team && sp.team !== 'all') filters.push({ field: 'team_id', op: 'eq', value: sp.team });
      if (sp.cycle && sp.cycle !== 'all') filters.push({ field: 'cycle_id', op: 'eq', value: sp.cycle });

      const [byAgent, byTeamSeries] = await Promise.all([
        fetchQuery({ metric: 'cost_by_agent', dimension: 'agent', filters, window: { last: window! } }),
        fetchQuery({ metric: 'agent_session_count', dimension: 'agent', filters, window: { last: window! } }),
      ]);

      if (byAgent.rows.length === 0) {
        return <EmptyState />;
      }
      // Anomaly highlights: server-side computation deferred to a future query API metric.
      // P1 lightweight implementation: pass byAgent and an empty anomalies array.
      return (
        <main className="mx-auto max-w-screen-2xl px-6 py-12 space-y-8">
          <header>
            <h1 className="text-[28px] font-semibold leading-tight">Cost</h1>
            <p className="mt-2 text-sm text-muted-foreground">Spend per agent, by team and cycle.</p>
          </header>
          {/* Top-bar filters: team / cycle / window — URL-stateful per UI-SPEC */}
          <FiltersBar selected={{ team: sp.team ?? 'all', cycle: sp.cycle ?? 'all', window: window! }} />
          <CostChart data={byAgent.rows} />
          <AgentsTable agentRows={byAgent.rows} sessionRows={byTeamSeries.rows} />
        </main>
      );
    }
    ```
