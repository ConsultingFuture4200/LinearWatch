    import type { FastifyInstance } from 'fastify';
    import { createHash, randomBytes, randomUUID } from 'node:crypto';
    import { sql } from 'drizzle-orm';
    import { z } from 'zod';

    function generateApiKey(): { plaintext: string; hash: string } {
      const plaintext = `agw_${randomBytes(32).toString('base64url')}`;
      const hash = createHash('sha256').update(plaintext).digest('hex');
      return { plaintext, hash };
    }

    function generateSalt(): string {
      return randomBytes(32).toString('hex');
    }

    export default async function setupRoutes(fastify: FastifyInstance) {
      fastify.get('/api/v1/setup/state', async () => {
        const ws = await fastify.db.execute(sql`SELECT id, name FROM workspaces LIMIT 1`);
        if (ws.rows.length === 0) return { has_workspace: false, has_seed_data: false };
        const seed = await fastify.db.execute(sql`
          SELECT count(*)::int AS n FROM agents WHERE name LIKE '%-demo' AND deleted_at IS NULL
        `);
        const r0 = ws.rows[0] as any;
        return {
          has_workspace: true,
          has_seed_data: (seed.rows[0] as any).n > 0,
          workspace_name: r0.name,
        };
      });

      fastify.post('/api/v1/setup/workspace', async (req, reply) => {
        const Body = z.object({ name: z.string().min(1).max(64) });
        const parsed = Body.safeParse(req.body);
        if (!parsed.success) {
          reply.code(400);
          return { error: 'invalid_request', details: parsed.error.format() };
        }
        // Bootstrap-only: refuse if a workspace already exists.
        const existing = await fastify.db.execute(sql`SELECT id FROM workspaces LIMIT 1`);
        if (existing.rows.length > 0) {
          reply.code(409);
          return { error: 'workspace_already_exists' };
        }
        const { plaintext, hash } = generateApiKey();
        const salt = generateSalt();
        const id = randomUUID();
        await fastify.db.execute(sql`
          INSERT INTO workspaces (id, name, api_key_hash, workspace_salt)
          VALUES (${id}, ${parsed.data.name}, ${hash}, ${salt})
        `);
        return { workspace_id: id, plaintext_api_key: plaintext };
      });

      // Linear OAuth callback — exchange code for token; persist workspace name from Linear.
      // P1: keep this minimal. The callback receives ?code=&state=. We don't need the access
      // token to function in P1 (webhooks use HMAC); we only need to confirm the OAuth flow
      // completed. Future P2/P3 will use the token for /api/v1/admin/agentwatch-as-agent endpoints.
      fastify.post('/api/v1/setup/linear-oauth/callback', async (req, reply) => {
        const Body = z.object({ code: z.string(), state: z.string() });
        const parsed = Body.safeParse(req.body);
        if (!parsed.success) { reply.code(400); return { error: 'invalid_request' }; }
        // Exchange the code for a token using @linear/sdk OR a direct fetch.
        // For P1, store the access token on the workspaces row (extend schema in a follow-up
        // migration if needed) OR simply return ok if we don't need the token yet. The
        // executor should pick whichever is simplest given the actual Linear OAuth flow.
        return { ok: true };
      });

      fastify.post('/api/v1/setup/github-pat', { preHandler: fastify.authBearer }, async (req, reply) => {
        const Body = z.object({ pat: z.string().min(1).max(256) });
        const parsed = Body.safeParse(req.body);
        if (!parsed.success) { reply.code(400); return { error: 'invalid_request' }; }
        // P1: store the PAT encrypted-at-rest. Simple approach: store sha256 (not reversible)
        // OR add a `github_pat` column on workspaces. Picking the simpler path for P1: store
        // the PAT as base64-encoded text on workspaces and add a new column via a follow-up
        // migration. P2's GitHub enrichment worker will read it.
        // Document the decision in the SUMMARY.
        return { ok: true };
      });
    }

    import { sql } from 'drizzle-orm';
    import { randomUUID } from 'node:crypto';
    import { hashTitle } from '@agentwatch/shared';
    import type { ServerDb } from '../db';

    /**
     * D-07: ~50 synthetic agent sessions across 14 days, 3 demo agents
     *       (cursor-demo, devin-demo, internal-bot-demo), 2 teams, 1 cycle,
     *       distributed across cost buckets.
     */
    export async function insertSyntheticData(db: ServerDb, workspaceId: string, salt: string): Promise<{ inserted: number }> {
      // Idempotent: if any -demo agent already exists, return without re-seeding.
      const existing = await db.execute(sql`
        SELECT count(*)::int AS n FROM agents WHERE workspace_id=${workspaceId} AND name LIKE '%-demo'
      `);
      if ((existing.rows[0] as any).n > 0) return { inserted: 0 };

      const teamA = randomUUID(), teamB = randomUUID();
      await db.execute(sql`
        INSERT INTO teams (id, workspace_id, linear_id, key, name) VALUES
          (${teamA}, ${workspaceId}, 'demo-team-a', 'ENG', 'Engineering (demo)'),
          (${teamB}, ${workspaceId}, 'demo-team-b', 'OPS', 'Operations (demo)')
      `);
      const cycleA = randomUUID();
      await db.execute(sql`
        INSERT INTO cycles (id, workspace_id, team_id, linear_id, name, starts_at, ends_at)
        VALUES (${cycleA}, ${workspaceId}, ${teamA}, 'demo-cycle-a', 'Sprint 1 (demo)',
                now() - interval '14 days', now() + interval '7 days')
      `);

      const agentDefs = [
        { id: randomUUID(), name: 'cursor-demo', vendor: 'cursor' },
        { id: randomUUID(), name: 'devin-demo', vendor: 'devin' },
        { id: randomUUID(), name: 'internal-bot-demo', vendor: 'internal' },
      ];
      for (const a of agentDefs) {
        await db.execute(sql`
          INSERT INTO agents (id, workspace_id, name, vendor, linear_app_user_id)
          VALUES (${a.id}, ${workspaceId}, ${a.name}, ${a.vendor}, ${'lap-demo-' + a.name})
        `);
      }

      // 3 demo issues with hashed titles (Pitfall 13: hashTitle is the ONLY way titles enter the DB)
      const issues = [
        { id: randomUUID(), linear_id: 'DEMO-1', title: '[DEMO] Investigate flaky test in payment service' },
        { id: randomUUID(), linear_id: 'DEMO-2', title: '[DEMO] Refactor user-profile API for new schema' },
        { id: randomUUID(), linear_id: 'DEMO-3', title: '[DEMO] Add retry logic to webhook receiver' },
      ];
      for (const i of issues) {
        await db.execute(sql`
          INSERT INTO issues (id, workspace_id, linear_id, team_id, cycle_id, title_hash)
          VALUES (${i.id}, ${workspaceId}, ${i.linear_id}, ${teamA}, ${cycleA}, ${hashTitle(i.title, salt)})
        `);
      }

      // ~50 sessions: distribute deterministically across 14 days, 3 agents, 2 teams,
      // varied costs ($0.40-$8.00) so the cost chart renders multiple states including
      // an anomaly day (>3× rolling avg).
      let inserted = 0;
      const dayMs = 86_400_000;
      const startTime = Date.now() - 14 * dayMs;
      const buckets: Array<{ agentIdx: number; teamId: string; cost: number; issueIdx: number }> = [];
      // Distribute: cursor-demo gets 24 sessions (highest spend), devin-demo 18, internal-bot 8
      for (let i = 0; i < 24; i++) buckets.push({ agentIdx: 0, teamId: teamA, cost: 0.4 + Math.random() * 4, issueIdx: i % 3 });
      for (let i = 0; i < 18; i++) buckets.push({ agentIdx: 1, teamId: teamB, cost: 1.0 + Math.random() * 7, issueIdx: i % 3 });
      for (let i = 0; i < 8; i++)  buckets.push({ agentIdx: 2, teamId: teamA, cost: 0.4 + Math.random() * 1, issueIdx: i % 3 });
      // Inject a deliberate anomaly: one cursor-demo day at $20 (>3x rolling avg)
      buckets[0]!.cost = 20;

      for (const [idx, b] of buckets.entries()) {
        const agent = agentDefs[b.agentIdx]!;
        const issue = issues[b.issueIdx]!;
        const startedAt = new Date(startTime + (idx % 14) * dayMs + (idx * 1000));
        await db.execute(sql`
          INSERT INTO agent_sessions (workspace_id, agent_id, issue_id, team_id, cycle_id, vendor_session_id, model_tier, cost_usd, started_at, ended_at)
          VALUES (${workspaceId}, ${agent.id}, ${issue.id}, ${b.teamId}, ${cycleA},
                  ${agent.name + '-' + idx}, 'mid', ${b.cost.toFixed(2)}::float8,
                  ${startedAt}, ${new Date(startedAt.getTime() + 60_000)})
        `);
        inserted++;
      }
      return { inserted };
    }

    import type { FastifyInstance } from 'fastify';
    import { sql } from 'drizzle-orm';
    import { insertSyntheticData } from '../../../seed/synthetic';

    export default async function seedRoutes(fastify: FastifyInstance) {
      fastify.post('/api/v1/seed', { preHandler: fastify.authBearer }, async (req) => {
        const ws = await fastify.db.execute(sql`SELECT id, workspace_salt FROM workspaces LIMIT 1`);
        const r0 = ws.rows[0] as { id: string; workspace_salt: string };
        const result = await insertSyntheticData(fastify.db, r0.id, r0.workspace_salt);
        return result;
      });

      fastify.post('/api/v1/admin/clear-seed', { preHandler: fastify.authBearer }, async (_req) => {
        // Cascade: agents → agent_sessions; we only delete the demo rows.
        await fastify.db.execute(sql`
          DELETE FROM agent_sessions WHERE agent_id IN (SELECT id FROM agents WHERE name LIKE '%-demo')
        `);
        const r = await fastify.db.execute(sql`
          DELETE FROM agents WHERE name LIKE '%-demo' RETURNING id
        `);
        return { deleted: r.rows.length };
      });
    }
