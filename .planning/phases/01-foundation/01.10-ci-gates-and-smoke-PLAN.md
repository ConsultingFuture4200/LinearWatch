        }

    `scripts/check-no-title-leak.sh`:
    ```bash
    #!/usr/bin/env bash
    # D-26 / Pitfall 13: issues schema must declare title_hash and no title column.
    set -euo pipefail
    SCHEMA=packages/db/src/schema/issues.ts
    if [ ! -f "$SCHEMA" ]; then
      echo "FAIL: $SCHEMA not found"; exit 1
    fi
    # The brand type must be present
    if ! grep -q 'titleHash\|title_hash\|TitleHash' "$SCHEMA"; then
      echo "FAIL: $SCHEMA missing TitleHash brand reference"; exit 1
    fi
    # No bare `title:` column declaration (must always be `titleHash` / `title_hash`)
    BAD=$(grep -nE "title\s*:\s*text\(|title\s*:\s*varchar\(|title\s*:\s*string" "$SCHEMA" || true)
    if [ -n "$BAD" ]; then
      echo "FAIL: $SCHEMA appears to declare a raw title column:"
      echo "$BAD"; exit 1
    fi
    # Migration SQL must not declare a title text column on issues
    if grep -nE "CREATE TABLE.*issues[^a-zA-Z_]" packages/db/migrations/*.sql 2>/dev/null \
       | xargs -I{} sh -c 'cat "{}"' 2>/dev/null \
       | grep -nE "^\s+title\s+text|^\s+title\s+varchar" >/dev/null; then
      echo "FAIL: migration SQL declares a title column on issues"; exit 1
    fi
    echo "OK: issues schema has title_hash; no title column."
    ```
